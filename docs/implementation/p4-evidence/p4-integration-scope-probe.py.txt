from pathlib import Path
from tempfile import TemporaryDirectory
from uuid import uuid4
import subprocess
from repo_catalog.adapters.git.native import GitNativeSource
from repo_catalog.adapters.git.notes import GitNotesSource
from repo_catalog.adapters.sqlite.git_core import GitCore
from repo_catalog.adapters.sqlite.notes import NotesProgress
from repo_catalog.adapters.sqlite.p1_coordination import WorkLedger, NotesWorkBudget
from repo_catalog.adapters.sqlite.p1_schema import initialize, connect
from repo_catalog.application.notes import NotesAcquisition
from repo_catalog.domain.p1_core import OID, Capture, encode
from repo_catalog.domain.p1_coordination import RequestIdentity, NotesLimits
for fmt in ('sha1','sha256'):
 with TemporaryDirectory() as temp:
  p=Path(temp); gitpath=p/'git'; gitpath.mkdir()
  def git(*args,data=None):
   return subprocess.run(['git','-C',str(gitpath),*args],input=data,check=True,capture_output=True).stdout
  git('init','--bare','--object-format='+fmt)
  def write(kind,data): return OID(fmt,git('hash-object','-w','-t',kind,'--stdin',data=data).strip().decode())
  note=write('blob',b'\0\xffnote\r\n'); target=OID(fmt,'a'*(40 if fmt=='sha1' else 64))
  tree=write('tree',b'100644 '+target.full_oid.encode()+b'\0'+bytes.fromhex(note.full_oid))
  actual_repo,actual_instance,wrong_repo,wrong_instance,origin=(str(uuid4()) for _ in range(5))
  cpath=p/'catalog-a'/'core.sqlite3';cpath.parent.mkdir(); initialize(cpath,'core',actual_repo,actual_instance)
  opath=p/'catalog-b'/'operational.sqlite3';opath.parent.mkdir();initialize(opath,'operational',wrong_repo,wrong_instance)
  core=GitCore(cpath);db=connect(opath,'operational');ledger=WorkLedger(db)
  req=RequestIdentity(origin,str(uuid4()));ledger.new_request(req,encode([wrong_repo,wrong_instance,'other-source','other-intent']),NotesLimits.work_limits()); attempt=str(uuid4());ledger.attempt(req,attempt);work=NotesWorkBudget(ledger,req,attempt,NotesLimits())
  app=NotesAcquisition(core,GitNotesSource(GitNativeSource(gitpath)),work,NotesProgress(work))
  cap=Capture(origin,str(uuid4()),actual_repo,'actual-source','refs/notes/a',tree)
  result=app.acquire_note(cap,target)
  print(fmt, 'cross_catalog_body_retained=',result.body_retained,'wrong_catalog_usage=',ledger.usage(req))
  assert result.body_retained
  assert core.read_note(cap,target,note)==b'\0\xffnote\r\n'
  core.close();db.close()
