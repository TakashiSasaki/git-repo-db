def deny_commit(c):
 original=c._authorize
 def authorizer(action,a,b,d,t):
  if action==sqlite3.SQLITE_TRANSACTION and a=='COMMIT':return sqlite3.SQLITE_DENY
  return original(action,a,b,d,t)
 c._db.set_authorizer(authorizer)
 return original

def test_real_commit_denial_atomic_group_and_body(actual):
 c,db,cap,target,note,source=actual
 inv=mapping_inventory(cap,source.metadata,Budget());original=deny_commit(c);work=Budget()
 with pytest.raises(sqlite3.DatabaseError):c.admit_notes_inventory(inv,work)
 c._db.set_authorizer(original)
 for table in ['notes_groups','note_roles','captures']:
  assert c.sql('SELECT count(*) FROM '+table)[0][0]==0
 assert work.counts['verification_bytes']>0
 assert not (db.parent/'p1-recovery-stop.json').exists()
 c.admit_notes_inventory(inv,Budget())
 proof=mapping_proof(cap,target,source.metadata,Budget());original=deny_commit(c)
 with pytest.raises(sqlite3.DatabaseError):c.admit_note(proof,note,str(uuid4()),Budget())
 c._db.set_authorizer(original)
 assert c.sql('SELECT count(*) FROM retained_notes')[0][0]==0
 assert len(c.notes_inventories(cap))==1
 assert c.notes_projection(cap,target).canonical==note.oid
 assert not (db.parent/'p1-recovery-stop.json').exists()

def test_replay_copies_original_report_and_capture_binding(actual):
 c,db,cap,target,note,source=actual
 other=OID(target.object_format,'b'*len(target.full_oid))
 report=NotesGroup(cap,'all-notes-mappings','complete',((target.full_oid,other.full_oid,None),),provenance('independent-source'))
 c.admit_notes_group(report); c.admit_notes_group(report)
 proof=mapping_proof(cap,target,source.metadata,Budget());copy=str(uuid4());work=Budget()
 c.admit_note(proof,note,copy,work);before=work.counts['verification_bytes'];c.admit_note(proof,note,copy,work)
 assert work.counts['verification_bytes']>before
 assert c.sql('SELECT count(*) FROM retained_notes')[0][0]==1
 c.admit_note(proof,note,str(uuid4()),Budget());assert c.sql('SELECT count(*) FROM retained_notes')[0][0]==2
 project=c.notes_projection(cap,target);assert project.reported_conflict and project.canonical==note.oid
 assert any(x.provenance==report.provenance for x in project.alternatives)
 assert c.read_note(cap,target,note.oid)==b'\x00\xff\r\n'
 for bad in [replace(cap,source='other'),replace(cap,ref='refs/notes/other'),replace(cap,root=other)]:
  with pytest.raises((CoreFailure,sqlite3.IntegrityError)):c.admit_capture(bad)
 assert c.sql('SELECT count(*) FROM captures')[0][0]==1
 assert c.read_note(cap,target,note.oid)==note.body
 for statement in ['UPDATE captures SET source=source','DELETE FROM notes_groups','INSERT OR REPLACE INTO notes_groups SELECT * FROM notes_groups']:
  with pytest.raises(sqlite3.DatabaseError):c.sql(statement)
