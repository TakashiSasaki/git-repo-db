def test_swallowed_sqlite_corruption_still_stops_before_other_methods(directory):
 class FaultConnection:
  def __init__(self, real):self.real=real;self.fault=True
  @property
  def in_transaction(self):return self.real.in_transaction
  def execute(self, sql, params=()):
   if self.fault and ' FROM resources ' in sql:
    self.fault=False
    error=sqlite3.DatabaseError('database disk image is malformed')
    error.sqlite_errorcode=sqlite3.SQLITE_CORRUPT
    raise error
   return self.real.execute(sql,params)
 db=connect(directory/'context.sqlite3','context');db.execute('BEGIN IMMEDIATE')
 try:
  with pytest.raises(CoordinationError, match='RECOVERY_REQUIRED'):
   with Context.for_transaction(FaultConnection(db),directory) as ctx:
    try:ctx.resources()
    except sqlite3.DatabaseError:pass
    assert (directory/'p1-recovery-stop.json').exists()
 finally:db.rollback();db.close()

@pytest.mark.parametrize('operation',['read','replay'])
def test_known_referenced_text_loss_stops_but_unknown_digest_does_not(directory,operation):
 with Context.transaction(directory) as ctx:
  ctx.admit_text(KEY,'original',FieldClock('provider.updated',1),PROVENANCE,10)
  digest=__import__('hashlib').sha256(b'original').hexdigest()
  with pytest.raises(ContextError) as error:ctx.read_text('b'*64)
  assert error.value.code=='TEXT_UNAVAILABLE'
  assert not (directory/'p1-recovery-stop.json').exists()
 db=connect(directory/'context.sqlite3','context')
 trigger=db.execute("SELECT sql FROM sqlite_schema WHERE name='text_bodies_no_delete'").fetchone()[0]
 db.execute('DROP TRIGGER text_bodies_no_delete');db.execute('PRAGMA foreign_keys=OFF');db.execute('DELETE FROM text_bodies');db.execute(trigger);db.commit();db.close()
 with pytest.raises(ContextError) as error:
  with getattr(Context, 'snapshot' if operation=='read' else 'transaction')(directory) as ctx:
   if operation=='read':ctx.read_text(digest)
   else:ctx.text('original')
 assert error.value.code=='AUTHORITATIVE_CORRUPTION'
 assert (directory/'p1-recovery-stop.json').exists()
