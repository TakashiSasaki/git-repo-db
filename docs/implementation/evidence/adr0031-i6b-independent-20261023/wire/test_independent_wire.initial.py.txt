def checksum(data):
    body={k:v for k,v in data.items() if k!='digest_sha256'}
    data['digest_sha256']=hashlib.sha256(json.dumps(body,ensure_ascii=False,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()
    return data

def package(ctx=None):

def refused(fn, value, expected=None):
    with pytest.raises(ExchangeError) as error: fn(value)
    if expected is not None: assert error.value.code==expected
    log=os.environ.get('WIRE_REFUSALS')
    if log:
        with open(log,'a') as stream: stream.write(json.dumps({'test':os.environ.get('PYTEST_CURRENT_TEST'),'type':type(error.value).__name__,'code':error.value.code,'scope':error.value.scope},ensure_ascii=True)+'\n')
    return error.value

def generic_relation(k, members, *, kind='x-members', observed=779, prov=PV, scope='{}'):

def collection(k,members,*,state='complete',observed=787,prov=PV,scope='{}'):

    row['collection_kind']=row.pop('relation_kind'); row['terminal_state']=state
    return row

def test_R03_outer_utf8_digest_recomputed_independently():
