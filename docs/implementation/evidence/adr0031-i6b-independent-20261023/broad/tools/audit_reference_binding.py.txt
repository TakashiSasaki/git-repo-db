import hashlib,json,subprocess
from pathlib import Path
ROOT=Path('/workspace/git-repo-db');REF=Path('/workspace/i6b-broad-work/reference149');OUT=ROOT/'docs/implementation/evidence/adr0031-i6b-independent-20261023/broad';HEAD='ea43a21fca1dbf8fa6dd1c254b28a96d5dd28d00'
rows=[]
for lane in ['full','diagnostic743']:
    before=json.loads((REF/lane/'binding-before.json').read_text());after=json.loads((REF/lane/'binding-after.json').read_text());assert before==after
    mismatch=[]
    for rel,digest in before['inputs'].items():
        raw=subprocess.check_output(['git','show',HEAD+':'+rel],cwd=ROOT)
        if hashlib.sha256(raw).hexdigest()!=digest:mismatch.append(rel)
    assert not mismatch
    rows.append({'lane':lane,'execution_head':before['head'],'execution_tree':before['tree'],'mapped_reference_head':HEAD,'mapped_reference_tree':subprocess.check_output(['git','rev-parse',HEAD+'^{tree}'],cwd=ROOT,text=True).strip(),'inputs_checked':len(before['inputs']),'each_input_matches_actual_reference_Git_blob':True,'before_after_binding_equal':True,'mismatches':mismatch})
(OUT/'reference149-input-binding-audit.json').write_text(json.dumps(rows,indent=2)+'\n');print(json.dumps(rows))
