def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def write(path, data):
    Path(path).write_text(json.dumps(data, indent=2) + "\n")


def run(name, command, cwd=SOURCE, env=None):
    started = time.time()
    with (OUT / (name + ".log")).open("w") as stream:
        result = subprocess.run(
            list(map(str, command)),
            cwd=cwd,
            env=env,
            stdout=stream,
            stderr=subprocess.STDOUT,
        )
    record = {
        "name": name,
        "command": list(map(str, command)),
        "cwd": str(cwd),
        "exit_code": result.returncode,
        "started_epoch": started,
        "elapsed_seconds": round(time.time() - started, 3),
        "source_head": COMMIT,
        "source_tree": TREE,
        "driver_sha256": sha(__file__),
    }
    records.append(record)
    write(OUT / "processes.json", records)
    print(json.dumps(record), flush=True)
    return record


def require(record):
    if record["exit_code"]:
        raise RuntimeError("unsuccessful process: " + record["name"])
    return record


def product_files(root):
    return {
        str(p.relative_to(root)): sha(p)
        for p in sorted(root.rglob("*"))
        if p.is_file() and "__pycache__" not in p.parts and p.suffix != ".pyc"
    }


def verify_wheel(artifact):
    with zipfile.ZipFile(artifact) as archive:
        names = archive.namelist()
        assert len(names) == len(set(names)), "duplicate wheel member"
        files = {n: archive.read(n) for n in names if not n.endswith("/")}
    assert not any(n.startswith(("tests/", "scripts/")) for n in files)
    record_name = next(n for n in files if n.endswith(".dist-info/RECORD"))
    rows = list(csv.reader(io.StringIO(files[record_name].decode())))
    assert collections.Counter(row[0] for row in rows) == collections.Counter(
        files.keys()
    )
    for name, digest, size in rows:
        if name == record_name:
            assert not digest and not size
        else:
            expected = (
                base64.urlsafe_b64encode(hashlib.sha256(files[name]).digest())
                .decode()
                .rstrip("=")
            )
            assert digest == "sha256=" + expected
            assert int(size) == len(files[name])
    return files

def pytest_lane(tag, label, targets, outside, python, env):
    label = tag + "-" + label
    arguments = [
        python,
        "-m",
        "pytest",
        "-q",
        "-p",
        "i6a_report_types",
        "-o",
        "addopts=--strict-markers",
        "-o",
        "junit_family=xunit1",
        "-c",
        outside / "pytest.ini",
        "--rootdir=" + str(outside),
        "--confcutdir=" + str(outside),
        *targets,
    ]
    env = dict(env, REPO_CATALOG_IMPORT_RECEIPT=str(OUT / (label + "-imports.json")), I6A_REPORTS_SIDECAR=str(OUT / (label + "-report-types.jsonl")))
    require(run(label + "-collect", arguments + ["--collect-only"], outside, dict(env, I6A_REPORTS_SIDECAR=str(OUT / (label + "-collection-report-types.jsonl")))))
    selected = evidence.nodes(OUT / (label + "-collect.log"))
    (OUT / (label + "-selected.txt")).write_text("".join(n + "\n" for n in selected))
    if label.endswith("-safety"):
        historical = [
            line
            for line in gzip.open(
                SOURCE
                / "docs/implementation/evidence/adr0031-i6a/author/distribution-v3/direct-selected.txt.gz",
                "rt",
            )
            .read()
            .splitlines()
            if line.startswith("tests/") and "::" in line
        ]
        cohorts = {
            "i4": [
                n
                for n in selected
                if n.startswith("tests/integration/p3/test_transfer_sql.py::")
            ],
            "i5": [
                n
                for n in selected
                if n.startswith("tests/integration/p3/test_transfer_admission.py::")
            ],
            "originals_m1": [
                n
                for n in selected
                if n.startswith(
                    "tests/integration/i6a/test_originals_revalidation.py::"
                )
            ],
        }
        write(
            OUT / (label + "-cohort-check.json"),
            {
                "historical_count": len(historical),
                "selected_count": len(selected),
                "missing_historical": sorted(set(historical) - set(selected)),
                "additional": sorted(set(selected) - set(historical)),
                "cohort_counts": {k: len(v) for k, v in cohorts.items()},
                "cohorts": cohorts,
            },
        )
        baseline = Path('/workspace/i6b-work/baseline-task-c/installed-v3/direct-safety-selected.txt').read_text().splitlines()
        assert set(baseline) <= set(selected), 'required historical safety nodes omitted'
        assert set(historical) <= set(selected), 'required earlier safety nodes omitted'
    elif label.endswith(('-authorization','-task-c')):
        lane = 'authorization' if label.endswith('-authorization') else 'task-c'
        baseline = Path('/workspace/i6b-work/baseline-task-c/installed-v3/direct-'+lane+'-selected.txt').read_text().splitlines()
        assert selected == baseline, 'reused historical probe node identities changed'
    else:
        assert selected, 'empty installed wire cohort'
    junit, journal = OUT / (label + ".xml"), OUT / (label + ".jsonl")
    record = run(
        label,
        arguments
        + [
            "--basetemp=" + str(WORK / (label + "-tmp")),
            "--junitxml=" + str(junit),
            "--journal=" + str(journal),
        ],
        outside,
        env,
    )
    result = {
        **record,
        **evidence.outcomes(junit),
        "reconciliation": evidence.reconciliation(selected, junit, journal),
    }
    write(OUT / (label + "-execution.json"), result)
    assert evidence.cohort_succeeded(result), "installed cohort unsuccessful: " + label
    return result

        except Exception as error:
            failures.append(
                {"form": tag, "exception": type(error).__name__, "message": str(error)}
            )
            (OUT / (tag + "-driver-failure.log")).write_text(traceback.format_exc())
    write(
        OUT / "summary.json",
        {
            "source_head": COMMIT,
            "source_tree": TREE,
            "processes": records,
            "results": results,
            "failures": failures,
            "full_acceptance": False,
            "independent_review": False,
        },
    )
    return int(bool(failures) or any(row["exit_code"] for row in records))


if __name__ == "__main__":
    raise SystemExit(main())
