"""Verify actual timestamped in-log CI receipt and exact feature/checkout/run binding."""

import argparse
import base64
import hashlib
import io
import json
import re
import zipfile
from pathlib import Path


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("log_zip", type=Path)
    parser.add_argument("output", type=Path)
    parser.add_argument("--feature", required=True)
    parser.add_argument("--tree", required=True)
    parser.add_argument("--base", required=True)
    parser.add_argument("--run", required=True)
    parser.add_argument("--attempt", required=True)
    parser.add_argument("--checkout", required=True)
    args = parser.parse_args()
    args.output.mkdir(parents=True, exist_ok=True)
    raw_log_zip = args.log_zip.read_bytes()
    with zipfile.ZipFile(io.BytesIO(raw_log_zip)) as logs:
        candidates = []
        for name in logs.namelist():
            if name.endswith("/"):
                continue
            text = logs.read(name).decode("utf-8")
            lines = []
            for line in text.splitlines():
                match = re.match(
                    r"^\d{4}-\d\d-\d\dT\S+Z\s+(CI_RECEIPTS_(?:BEGIN|DATA|END)_V1 .*)$",
                    line,
                )
                if match:
                    lines.append(match.group(1))
            if lines:
                candidates.append((name, lines))
    assert candidates, "no actual timestamped receipt markers"
    # The API archive may expose step and combined logs for the same job. Every
    # occurrence must agree; none is trusted just because it is encountered first.
    receipts = []
    for name, lines in candidates:
        begin = [line for line in lines if line.startswith("CI_RECEIPTS_BEGIN_V1 ")]
        end = [line for line in lines if line.startswith("CI_RECEIPTS_END_V1 ")]
        assert len(begin) == len(end) == 1, (name, "missing/duplicate markers")
        _, digest, size = begin[0].split()
        assert end[0].split()[1] == digest
        assert lines[0] == begin[0] and lines[-1] == end[0]
        encoded = "".join(
            line.split()[1]
            for line in lines[1:-1]
            if line.startswith("CI_RECEIPTS_DATA_V1 ")
        )
        body = base64.b64decode(encoded, validate=True)
        assert len(body) == int(size)
        assert hashlib.sha256(body).hexdigest() == digest
        receipts.append(body)
    assert all(body == receipts[0] for body in receipts)
    body = receipts[0]
    with zipfile.ZipFile(io.BytesIO(body)) as archive:
        names = archive.namelist()
        assert len(names) == len(set(names))
        assert all(
            "/" not in name and "\\" not in name and name not in (".", "..")
            for name in names
        )
        files = {name: archive.read(name) for name in names}
    index = json.loads(files["receipt-index.json"])
    assert set(index) == set(files) - {"receipt-index.json"}
    for name, record in index.items():
        assert len(files[name]) == record["bytes"], name
        assert hashlib.sha256(files[name]).hexdigest() == record["sha256"], name
    binding = json.loads(files["receipt-binding.json"])
    for key, expected in [
        ("feature_sha", args.feature),
        ("tree", args.tree),
        ("head", args.checkout),
        ("run_id", args.run),
        ("run_attempt", args.attempt),
    ]:
        assert str(binding[key]) == expected, (key, binding[key], expected)
    manifest = json.loads(files["validation-manifest.json"])
    assert manifest["acceptance_status"] == "development_checks_passed"
    assert manifest["full_acceptance"] is False
    for name, content in files.items():
        (args.output / name).write_bytes(content)
    (args.output / "verified-receipt.zip").write_bytes(body)
    summary = {
        "receipt_bytes": len(body),
        "receipt_sha256": hashlib.sha256(body).hexdigest(),
        "log_zip_bytes": len(raw_log_zip),
        "log_zip_sha256": hashlib.sha256(raw_log_zip).hexdigest(),
        "indexed_payloads": len(index),
        "binding": binding,
        "manifest": manifest,
        "expected_base": args.base,
        "expected_feature": args.feature,
        "expected_tree": args.tree,
        "scope": "Development CI selected checks only; not full product or release acceptance.",
    }
    (args.output / "verification.json").write_text(json.dumps(summary, indent=2) + "\n")
    print(
        json.dumps(
            {key: value for key, value in summary.items() if key != "manifest"},
            indent=2,
        )
    )


if __name__ == "__main__":
    main()
