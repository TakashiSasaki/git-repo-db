def regex(pattern):
 assert not any(c in pattern for c in '![]?+'),pattern
 text='';i=0
 while i<len(pattern):
  if pattern[i:i+3]=='**/': text+='(?:.*/)?';i+=3
  elif pattern[i:i+2]=='**': text+='.*';i+=2
  elif pattern[i]=='*':text+='[^/]*';i+=1
  else:text+=re.escape(pattern[i]);i+=1
 return re.compile('^'+text+'$')
