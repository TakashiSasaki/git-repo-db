def case(name, function):
    entry = {"case": name}
    cases.append(entry)
    try:
        entry["observed"] = function()
        entry["outcome"] = "passed"
    except Exception as error:
        entry.update(outcome="failed", error=type(error).__name__, message=str(error))


def refuses(function, error_type=ValueError, contains=None):
    try:
        function()
    except error_type as error:
        if contains is not None:
            assert contains in str(error), str(error)
        return {"error": type(error).__name__, "message": str(error)}
    raise AssertionError("Expected refusal did not occur")


def distinct_native():
    values = ["731", ["731"], ["node", "731"], ["decimal", "731"], '["node","731"]']
    keys = [ResourceKey.from_native(S, "issue", value) for value in values]
    assert len(set(keys)) == len(values)
    for key in keys:
        assert ResourceKey(*key.coordinates()) == key
    return [key.provider_resource_id for key in keys]

def factory_contract():
    p = ResourceKey.from_native(S, "pull_request", "PR1")
    scoped = ResourceKey.document(p, "document", "body", "D1")
    native = ["pull_request", p.provider_resource_id, "body", "D1"]
    explicit = ResourceKey.global_document(S, native)
    generic = ResourceKey.from_native(S, "document", native)
    assert explicit == generic == scoped
    assert hash(explicit) == hash(scoped)
    assert dataclasses.asdict(explicit) == dataclasses.asdict(scoped)
    assert list(dataclasses.asdict(explicit)) == ["service_instance_uuidv4", "resource_kind", "provider_resource_id"]
    return [refuses(key.scoped_document_coordinates, contains="global") for key in (explicit, generic)]

def github_namespaces():
    decimal = github_resource_id({"id": 731}, graphql=False)
    node = github_resource_id({"id": "731"}, graphql=True)
    assert parse_identity(decimal) == "731"
    assert parse_identity(node) == ["github:node_id", "731"]
    assert decimal != node
    assert github_timeline_input_id(731, graphql=True) == decimal
    assert github_timeline_input_id("731", graphql=True) == node
    assert github_repository_id({"id": "731", "databaseId": 731}, graphql=True) == node
    return {"decimal": decimal, "node": node}


case("github-database-and-node-namespaces-never-double-encoded", github_namespaces)


def github_scoped_nodes():
    unit = normalize_pull_request(S, {"id": "PR_node", "body": "body"}, 1, PROV, graphql=True)
    docs = [obs.resource for obs in unit.observations if obs.resource != unit.primary_resource]
