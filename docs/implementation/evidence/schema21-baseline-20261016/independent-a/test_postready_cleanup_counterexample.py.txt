"""Independent real SQLite reader race after valid generation promotion."""
import sqlite3

import pytest

from repo_catalog.adapters.sqlite import index
from tests.integration.test_catalog3_current_queries import current_catalog as current_catalog


def test_real_search_reader_lock_after_promotion_reports_failed_rebuild(current_catalog, monkeypatch):
    c = current_catalog
    old = c.candidate('issue', '1', 'old body')
    assert c.admit(old).status == 'accepted'
    index.rebuild(c.store, 'issue')
    assert c.admit({**old, 'body': 'new body', 'provider_updated_at_us': 3, 'observed_at_us': 4}).status == 'accepted'
    original = index._prune_obsolete_documents
    calls = 0
    blocker = None
    def pin_reader_after_ready(search, kind, keys):
        nonlocal calls, blocker
        calls += 1
        if calls == 2:
            blocker = sqlite3.connect(search.db_path, isolation_level=None)
            blocker.execute('BEGIN')
            blocker.execute('SELECT * FROM search_documents').fetchall()
            search.execute('PRAGMA busy_timeout=0')
        return original(search, kind, keys)
    monkeypatch.setattr(index, '_prune_obsolete_documents', pin_reader_after_ready)
    try:
        with pytest.raises(sqlite3.OperationalError, match='locked'):
            index.rebuild(c.store, 'issue')
    finally:
        if blocker is not None:
            blocker.close()
    with c.store.open_search(readonly=True) as s:
        ready = s.one("SELECT * FROM index_generations WHERE state='ready'")
        assert s.one(f"SELECT count(*) FROM {ready['table_name']} WHERE {ready['table_name']} MATCH 'new'")[0] == 1
        assert s.one(f"SELECT count(*) FROM {ready['table_name']} WHERE {ready['table_name']} MATCH 'old'")[0] == 0
    assert c.store.operational.one('SELECT status FROM search_rebuild_metrics ORDER BY rebuild_id DESC')[0] == 'failed'
