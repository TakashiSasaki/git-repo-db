def git(*args):
    return subprocess.check_output(['git',*args]).decode().strip()

def content(ref,path):
    return subprocess.check_output(['git','show',f'{ref}:{path}'])

def merge(first,second,label):
    result=subprocess.run(['git','merge-tree','--write-tree',first,second],text=True,capture_output=True)
    assert result.returncode==0,(label,result.stdout,result.stderr)
    tree=result.stdout.strip()
    commit=subprocess.check_output(['git','commit-tree',tree,'-p',first,'-p',second],input=(label+' local nonremote simulation\n').encode(),env=ENV).decode().strip()
    return {'label':label,'first_parent':first,'second_parent':second,'merge_base':git('merge-base',first,second),'tree':tree,'synthetic_commit':commit,'conflicts':False}
