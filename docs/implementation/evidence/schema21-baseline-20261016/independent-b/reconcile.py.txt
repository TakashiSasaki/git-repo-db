"""Reconcile Reviewer B actual JUnit, selected-node collections and journals."""
import collections
import gzip
import hashlib
import json
import subprocess
import xml.etree.ElementTree as ET
from pathlib import Path

BASE = Path('/workspace/reviewer-b-evidence')

def key(node):
    parts = node.split('::')
    return (parts[0][:-3].replace('/', '.') + ('.' + '.'.join(parts[1:-1]) if len(parts) > 2 else ''), parts[-1])

def check(directory, expected, *, junit='junit.xml', journal='journal.jsonl', collection='collect.log', selected=None):
    folder = BASE / directory
    cases = list(ET.parse(folder/junit).iter('testcase'))
    records = [json.loads(line) for line in (folder/journal).read_text().splitlines()]
    starts = [r['nodeid'] for r in records if r['event'] == 'start']
    nodes = selected if selected is not None else [line.strip() for line in (folder/collection).read_text().splitlines() if line.startswith('tests/') and '::' in line]
    keys = [(case.attrib['classname'], case.attrib['name']) for case in cases]
    assert len(nodes) == len(set(nodes)) == len(cases) == len(set(keys)) == len(starts) == len(set(starts)) == expected
    assert set(nodes) == set(starts)
    assert set(keys) == {key(node) for node in nodes}
    assert not any(case.find(tag) is not None for case in cases for tag in ('failure', 'error', 'skipped'))
    phases = [r for r in records if r['event'] == 'phase']
    assert all(r['outcome'] == 'passed' for r in phases)
    assert collections.Counter((r['nodeid'], r['phase']) for r in phases) == {(node, phase):1 for node in nodes for phase in ('setup', 'call', 'teardown')}
    finish = [r for r in records if r['event'] == 'finish']
    assert len(finish) == 1 and finish[0]['exitstatus'] == 0
    return {'selected': expected, 'junit': len(cases), 'journal_unique_starts': len(set(starts)), 'passed_phases': len(phases), 'exitstatus': 0, 'exactly_once': True, 'passed_node_ids': nodes}

summary = {'lifecycle': check('lifecycle-713df96', 20), 'combined_public_cli': check('p7-comparison-98c818a', 2, junit='public-lifecycle-junit.xml', journal='public-lifecycle-journal.jsonl', selected=['tests/e2e/test_recovery.py::test_kill_before_publish', 'tests/e2e/test_jobs.py::test_cancel_resume'])}
if (BASE/'p7-comparison-98c818a/junit.xml').exists():
    summary['combined_p7_and_compatibility'] = check('p7-comparison-98c818a', 304)
    nodes = summary['combined_p7_and_compatibility']['passed_node_ids']
    original = subprocess.check_output(['git', 'show', 'e956cd50b1f25bd95427301cced4741bb68e24f8:docs/implementation/evidence/p7-hardening-20261016/final/independent/d/wheel-collect.log.gz'], cwd='/workspace/review-b-p7')
    original_nodes = {line.strip() for line in gzip.decompress(original).decode().splitlines() if line.startswith('tests/') and '::' in line}
    actual_p7 = {node for node in nodes if node.startswith('tests/integration/p7/')}
    assert actual_p7 == original_nodes and len(actual_p7) == 255
    summary['combined_p7_and_compatibility']['p7_nodes_equal_original_frozen_installed_set'] = True
    summary['combined_p7_and_compatibility']['scope_counts'] = {'P7':255, 'P2-P3 compatibility':20, 'P5-P6 compatibility':29}
(BASE/'reconciliation.json').write_text(json.dumps(summary, indent=2)+'\n')
print(json.dumps({scope:{k:v for k,v in result.items() if k != 'passed_node_ids'} for scope,result in summary.items()}, indent=2))
