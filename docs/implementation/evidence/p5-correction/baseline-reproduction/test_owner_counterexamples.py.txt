def test_native_external_io_outside_write_transaction(tmp_path, monkeypatch):
    cat=tmp_path/'cat'; repo,_=init_catalog_stores(cat)
    oids=build_sample_git_repo(tmp_path/'git')
    source=GitNativeSource(tmp_path/'git')
    original=source.metadata
    seen=[]
    def checked(oid,*args,**kwargs):
        db=sqlite3.connect(cat/'core.sqlite3',timeout=0)
        try:
            db.execute('BEGIN IMMEDIATE'); db.rollback(); seen.append('free')
        except sqlite3.OperationalError:
            seen.append('locked')
        finally: db.close()
        return original(oid,*args,**kwargs)
    monkeypatch.setattr(source,'metadata',checked)
    req=GitNativeAcquisitionRequest(repo,str(uuid4()),str(uuid4()),'synthetic',1,select_oids=(oids['commit1'],),include_refs=False)
    PairedAcquisitionCoordinator(cat).admit_paired(git_source=source,git_request=req)
    assert seen and set(seen)=={'free'}, seen

def test_retry_preserves_physical_copy(tmp_path):
    cat=tmp_path/'cat'; repo,_=init_catalog_stores(cat)
    oids=build_sample_git_repo(tmp_path/'git')
    req=GitNativeAcquisitionRequest(repo,str(uuid4()),str(uuid4()),'synthetic',1,select_oids=(oids['commit1'],),include_refs=False,retain_bytes=True)
    acq=GitAcquisition(cat/'core.sqlite3',operational=cat)
    acq.acquire_from_native(GitNativeSource(tmp_path/'git'),req)
    with sqlite3.connect(cat/'core.sqlite3') as db: before=db.execute('SELECT count(*) FROM retained_metadata').fetchone()[0]
    acq.acquire_from_native(GitNativeSource(tmp_path/'git'),req)
    with sqlite3.connect(cat/'core.sqlite3') as db: after=db.execute('SELECT count(*) FROM retained_metadata').fetchone()[0]
    assert before==after


def test_missing_request_after_operational_loss_refuses_resume(tmp_path):
    from repo_catalog.domain.p1_coordination import CoordinationError
    cat=tmp_path/'cat'; repo,inst=init_catalog_stores(cat)
    req=RequestIdentity(str(uuid4()),str(uuid4()))
    with OperationalAcquisitionLedger(cat) as ledger:
        _,_,budget,_=ledger.create_budget(scope=['synthetic'],request=req)
        budget.require('input_bytes',5); budget.charge('input_bytes',5)
    for p in cat.glob('operational.sqlite3*'): p.unlink()
    initialize(cat/'operational.sqlite3','operational',repo,inst)
    with OperationalAcquisitionLedger(cat) as ledger:
        with pytest.raises(CoordinationError):
            ledger.create_budget(scope=['synthetic'],request=req)


def test_exchange_retains_partial_admission_outcome(tmp_path,monkeypatch):
    from repo_catalog.application.exchange import ExchangeService
    from repo_catalog.domain.exchange import empty_core_section,empty_context_section
    from repo_catalog.domain.p1_coordination import AdmissionFailure,PairAdmission
    cat=tmp_path/'cat'; repo,inst=init_catalog_stores(cat)
    core=empty_core_section(); core['git_objects'].append({'object_format':'sha1','full_oid':'a'*40})
    p=ExchangePackage(str(uuid4()),1,str(uuid4()),repo,core=core,context=empty_context_section())
    service=ExchangeService(cat)
    def bad(*args,**kwargs): raise ValueError('synthetic context failure')
    monkeypatch.setattr(service,'_import_context_into',bad)
    with pytest.raises(AdmissionFailure) as err:
        service.import_package(p)
    assert err.value.outcome==PairAdmission(True,False)


def test_exchange_export_uses_one_paired_snapshot(tmp_path,monkeypatch):
    from repo_catalog.application.exchange import ExchangeService
    from repo_catalog.adapters.sqlite.production_context import Context
    from repo_catalog.domain.p1_context import ResourceKey
    from repo_catalog.domain.p1_core import OID
    cat=tmp_path/'cat'; repo,inst=init_catalog_stores(cat)
    service=ExchangeService(cat); original=service._export_core
    oid=OID('sha1','b'*40); svc=str(uuid4()); rk=ResourceKey(svc,'issue','1')
    def racing(*args,**kwargs):
        result=original(*args,**kwargs)
        def core(cap): cap.admit_object(oid)
        def ctx(cap):
            cap.provider(svc,'github','{}'); cap.resource(rk)
            cap.link_core(rk,'head',*oid.key,'{"evidence":"synthetic","source":"synthetic"}',lambda *_: True)
        PairedAcquisitionCoordinator(cat).admit_paired(core_fn=core,context_fn=ctx)
        return result
    monkeypatch.setattr(service,'_export_core',racing)
    p=service.export_package(exported_at_us=1)
    core_oids={i['full_oid'] for i in p.core['git_objects']}
    context_oids={i['full_oid'] for i in p.context['core_references']}
    assert context_oids <= core_oids, (context_oids,core_oids)
