def command(args,name):
 start=time.time();r=subprocess.run(args,text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
 (out/(name+'.txt')).write_text(r.stdout)
 record={'argv':args,'cwd':str(root),'exit_code':r.returncode,'elapsed_seconds':time.time()-start}
 commands.append(record);(out/'commands.json').write_text(json.dumps(commands,indent=2))
 print(name,r.returncode,flush=True)
 return r
