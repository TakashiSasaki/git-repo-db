WORKER=r'''

def worker(catalog,salvage,cut='normal'):
    return subprocess.Popen([sys.executable,'-c',WORKER,str(catalog),str(salvage),cut],stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)

def initialize_catalog(root):
    catalog=root/str(uuid4());catalog.mkdir()
    for role in ('core','context','operational','search'):
        initialize(catalog/f'{role}.sqlite3',role,REPO,INSTANCE)
    return catalog

def state(catalog):
    return json.loads((catalog/'p1-recovery-request-independent.json').read_text())

def count(catalog):
    with sqlite3.connect(catalog/'context.sqlite3') as db:
        return db.execute('SELECT count(*) FROM text_bodies').fetchone()[0]

def collect(process):
    out,err=process.communicate(timeout=20)
    return process.returncode,json.loads(out) if out else {},err

def main():
    results=[]
    with tempfile.TemporaryDirectory(prefix='reviewer-c-process-') as td:
        root=Path(td)
        paths=[]
        for text in ('independent process A','independent process B'):
            path=root/(str(uuid4())+'.jsonl')
            path.write_text(json.dumps({'candidate_type':'context_document','extraction_status':'clean','payload':{'text_value':text}})+'\n')
            paths.append(path)
        cat=initialize_catalog(root)
        procs=[worker(cat,path) for path in paths]
        outcomes=[collect(p) for p in procs]
        results.append({'probe':'concurrent_same_request','pass':all(r[0]==0 for r in outcomes) and count(cat)==1 and state(cat)['cumulative_usage']['input_records']==1,'results':outcomes,'count':count(cat),'usage':state(cat)['cumulative_usage']})
        for cut in ('reservation','admission'):
            cat=initialize_catalog(root)
            result=collect(worker(cat,paths[0],cut))
            reserved=json.loads((cat/'p1-recovery-request-independent.reserved.json').read_text())['cumulative_usage']
            count_before=count(cat)
            for role in ('operational','search'):
                (cat/f'{role}.sqlite3').unlink()
            retries=[collect(worker(cat,paths[0])) for _ in range(3)]
            usage=state(cat)['cumulative_usage']
            passed=result[0]==-signal.SIGKILL and sum(reserved.values())>0 and all(r[0]==0 for r in retries) and all(usage[k]>=v for k,v in reserved.items()) and count(cat)==1
            results.append({'probe':'hard_kill_'+cut,'pass':passed,'exit':result[0],'count_before_retry':count_before,'final_count':count(cat),'reserved':reserved,'final_usage':usage,'retry_exits':[r[0] for r in retries]})
        for damage in ('report_delete','report_truncate','report_bool_usage','checkpoint_delete','checkpoint_conflict','registry_delete','marker_delete'):
            cat=initialize_catalog(root)
            assert collect(worker(cat,paths[0]))[0]==0
            report=cat/'p1-recovery-request-independent.json'
            checkpoint=cat/'p1-recovery-request-independent.reserved.json'
            if damage=='report_delete':report.unlink()
            elif damage=='report_truncate':report.write_text('{"version":')
            elif damage=='report_bool_usage':
                val=state(cat);val['cumulative_usage']['input_records']=True;report.write_text(json.dumps(val))
            elif damage=='checkpoint_delete':checkpoint.unlink()
            elif damage=='checkpoint_conflict':
                val=json.loads(checkpoint.read_text());val['request_id']='foreign';checkpoint.write_text(json.dumps(val))
            elif damage=='registry_delete':(cat/'p1-recovery-requests.json').unlink()
            else:(cat/'p1-recovery-requests.initialized').unlink()
            result=collect(worker(cat,paths[1]))
            results.append({'probe':damage,'pass':result[0]!=0 and 'RecoveryError' in result[2] and count(cat)==1,'exit':result[0],'error_tail':result[2].splitlines()[-1],'count':count(cat)})
    print(json.dumps(results,indent=2))
    assert all(row['pass'] for row in results)
if __name__=='__main__':main()
