from pathlib import Path
import tempfile,os
from repo_catalog.adapters.sqlite.p1_schema import initialize
import repo_catalog.adapters.sqlite.recovery as module
R='11111111-1111-4111-8111-111111111111'; I='22222222-2222-4222-8222-222222222222'
p=Path(tempfile.mkdtemp());catalog=p/'catalog';catalog.mkdir();previous=p/'previous';previous.mkdir();(p/'active').symlink_to(previous)
for role in ['core','context']:initialize(catalog/f'{role}.sqlite3',role,R,I)
identity=p.stat();original=module.os.fsync;swapped=False
def swap(fd):
 global swapped
 original(fd);info=os.fstat(fd)
 if (info.st_dev,info.st_ino)==(identity.st_dev,identity.st_ino) and (p/'active').resolve()==catalog and not swapped:
  swapped=True;catalog.rename(p/'validated');catalog.mkdir();(catalog/'core.sqlite3').write_bytes(b'UNVALIDATED')
module.os.fsync=swap
try:
 module.SQLiteRecoveryAdapter().activate(catalog,p/'active')
 print('UNEXPECTED_SUCCESS')
except Exception as e:
 print(type(e).__name__,str(e),'PREVIOUS_LOCATOR_RESTORED',(p/'active').resolve()==previous)
