from pathlib import Path
import tempfile
from repo_catalog.adapters.sqlite.p1_schema import initialize
from repo_catalog.adapters.sqlite.recovery import SQLiteRecoveryAdapter,is_catalog_stopped
R='11111111-1111-4111-8111-111111111111'; I='22222222-2222-4222-8222-222222222222'; J='33333333-3333-4333-8333-333333333333'
p=Path(tempfile.mkdtemp()); catalog=p/'catalog';catalog.mkdir()
initialize(catalog/'core.sqlite3','core',R,I)

try: SQLiteRecoveryAdapter().create_consistent_backup(catalog,p/'backup')
