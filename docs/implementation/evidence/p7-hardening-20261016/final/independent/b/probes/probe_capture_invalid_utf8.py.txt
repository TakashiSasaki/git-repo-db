from pathlib import Path
import tempfile,sqlite3,json
from repo_catalog.adapters.sqlite.p1_schema import initialize,register_functions
from repo_catalog.adapters.sqlite.recovery import SQLiteRecoveryAdapter,is_catalog_stopped
R='11111111-1111-4111-8111-111111111111';I='22222222-2222-4222-8222-222222222222';C='33333333-3333-4333-8333-333333333333'
p=Path(tempfile.mkdtemp());catalog=p/'catalog';catalog.mkdir()
for role in ['core','context']:initialize(catalog/f'{role}.sqlite3',role,R,I)
c=sqlite3.connect(catalog/'core.sqlite3');register_functions(c)
c.execute('INSERT INTO git_objects VALUES(?,?)',('sha1','1'*40));c.execute('INSERT INTO captures VALUES(?,?,?,?,?,?,?)',(R,C,R,'source',json.dumps(['native-ref',b'refs/notes/test'.hex()],separators=(',',':')),'sha1','1'*40));c.commit()
triggers=c.execute("SELECT name,sql FROM sqlite_schema WHERE type='trigger' AND tbl_name='captures' AND sql LIKE '%BEFORE UPDATE%'").fetchall()
for name,sql in triggers:c.execute('DROP TRIGGER '+name)
c.execute('PRAGMA ignore_check_constraints=ON');c.execute("UPDATE captures SET ref=CAST(x'ff' AS TEXT)")
for name,sql in triggers:c.execute(sql)
c.commit(); print('ROWS',c.execute('SELECT hex(ref) FROM captures').fetchall());c.close()


for uri in [str(catalog/'core.sqlite3'),'file:'+str(catalog/'core.sqlite3')+'?mode=ro']:
 cc=sqlite3.connect(uri,uri=True);register_functions(cc)
 try:print('PRAGMA',uri,cc.execute('PRAGMA integrity_check').fetchall())
 except Exception as error:print('PRAGMA_ERROR',uri,type(error).__name__,error)
 cc.close()
print('MODULE',__import__('repo_catalog.adapters.sqlite.recovery',fromlist=['x']).__file__)
try:
 receipt=SQLiteRecoveryAdapter().create_consistent_backup(catalog,p/'backup');print('UNEXPECTED_BACKUP_SUCCESS',receipt.verified_clean);c=sqlite3.connect(p/'backup/stores/core.sqlite3'); print('BACKUP_ROWS',c.execute('SELECT hex(ref) FROM captures').fetchall());c.close()
except Exception as e:print(type(e).__name__,str(e),'STOPPED',is_catalog_stopped(catalog));print('CAUSE',repr(e.__cause__))
