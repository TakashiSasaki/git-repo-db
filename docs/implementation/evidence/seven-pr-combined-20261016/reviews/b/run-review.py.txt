import datetime, json, os, subprocess, sys, time
from pathlib import Path
root=Path('/workspace/review-b'); out=Path('/workspace/integration-acceptance/reviews/b')
name=sys.argv[1]; argv=sys.argv[2:]; env=os.environ.copy(); overrides={'TMPDIR':str(out/'tmp'),'PATH':str(root/'.venv/bin')+':'+env['PATH']}; env.update(overrides)
start=datetime.datetime.now(datetime.timezone.utc).isoformat(); tick=time.monotonic()
with (out/(name+'.log')).open('wb') as log:
 p=subprocess.run(argv,cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT)
r=dict(command=argv,cwd=str(root),environment_overrides=overrides,start=start,end=datetime.datetime.now(datetime.timezone.utc).isoformat(),elapsed_seconds=time.monotonic()-tick,exit_code=p.returncode,head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),tree=subprocess.check_output(['git','rev-parse','HEAD^{tree}'],cwd=root,text=True).strip())
(out/(name+'.execution.json')).write_text(json.dumps(r,indent=2)+'\n');print(json.dumps(r));sys.exit(p.returncode)
